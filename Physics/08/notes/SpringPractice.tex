  \begin{questions}

    \question
    A certain spring stretches {\bf 5.2~cm} when {\bf 130~g} are hung from it.  
    
    \begin{parts}
      \part What is the spring constant? \vspace{\stretch{1}}
      \part If it was allowed to oscillate, what would its period be? \vspace{\stretch{1}}
    \end{parts}

    \begin{solution}
      24.5 N/m
    \end{solution}


  \question
    What mass must you attach to a spring of constant 300~N/m in order for it to make 20 oscillations in 5 seconds?

    \begin{solution}[\stretch{2}]
      0.18 kg
    \end{solution}
