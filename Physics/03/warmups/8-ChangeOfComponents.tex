\def\content{
A crossbow is fired at an angle with an initial x-velocity of 40 m/s and an initial y-velocity of 30 m/s. The total initial velocity of the arrow is 50 m/s.

\hfill 

\noindent
\begin{tikzpicture}[
    every node/.append style={font=\footnotesize},
    scale=0.8
  ]

    \def\velx{40}
    \def\vely{30}

    \def\apexx{6.1}
    \def\apexy{2}
    \def\vx{\velx/20}
    \def\vy{\vely/20}
    \def\bnk{\fillin[][5em]}

    \draw (-1,-0.7) 
      -- ++(1,0) 
      -- ++(2*\apexx,0) 
      -- ++(1,0);

    \draw[dashed,red] (\apexx,\apexy) 
      coordinate (apex) parabola (0,0);
    \draw[dashed,red] (apex) parabola 
      (2*\apexx,0) coordinate (end);
  
    \draw[ball color=purple] (0,0) circle (0.2);
    \draw[ball color=purple] (apex) circle (0.2);
    \draw[ball color=purple] (end) circle (0.2);

    \draw[dashed, blue] (0,0) -- (\vx, 0)
      node[midway,below] {$v_x=\velx$~m/s}
      -- (\vx, \vy)
      node[midway,rotate=90,below] {$v_y=\vely$~m/s};
    \draw[ultra thick, blue, -Stealth] (0,0) -- (\vx, \vy)
      node[midway,sloped,above] {$v_R=50$~m/s};


  \end{tikzpicture}

  \hfill

\begin{parts}
  \part What is the total speed of the arrow at the top of its motion? How do you know? \vs 
  \part What is the total speed of the arrow just before it hits the ground? How do you know? \vs 
\end{parts}
}
