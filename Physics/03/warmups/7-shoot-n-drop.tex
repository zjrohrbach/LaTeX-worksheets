\def\content{

Consider two identical objects. Object A is dropped from a height of 1.2 m. Object B is launched horizontally (at 7 m/s) from the same height.  You may assume that there is no air resistance.

\hfill

\begin{tikzpicture}
  \begin{scope}
    \node[ball color=red,circle,white] at (-1,3) (A) {A};
    \node[ball color=red,circle,white] at (1,3) (B) {B};
    \draw[dashed] (A.south) ++(0,-.1) -- (-1,0.1);
    \draw[dashed] (B.east) ++(.1,-.1) parabola (5,0.1);
    \draw[fill=gray] (B) ++(0,-.44) rectangle (-.6,0);
    \draw (-3,0) -- (6,0);
    \draw[-Stealth, very thick, purple] (B) -- +(1.5,0) node[above] {$v_i=\SI{7}{m/s}$};
    \draw[|<->|] (-2,0) coordinate (bottom) -- (bottom |- A) node[midway, fill=white] {1.2 m};
  \end{scope}
\end{tikzpicture}

\hfill 

\begin{parts}
  \part Make a T-chart for both objects. 

  \begin{tikzpicture}
    \begin{scope}
      \node at (0,4) {\bf Ball A};
      \draw (-2,3) -- (2,3);
      \draw (0,3.5) -- (0,0);
      \node at (-1,3.25) {$x$-direction};
      \node at (1,3.25) {$y$-direction};
    \end{scope}

    \begin{scope}[shift={(6,0)}]
      \node at (0,4) {\bf Ball B};
      \draw (-2,3) -- (2,3);
      \draw (0,3.5) -- (0,0);
      \node at (-1,3.25) {$x$-direction};
      \node at (1,3.25) {$y$-direction};
    \end{scope}
  \end{tikzpicture}

  \part Which object will hit the ground first? (\emph{You don't need to actually calculate anything; just look at your T-charts}) \vs 
  \part Which object will be traveling faster when it hits the ground? \vs
\end{parts}


}
