\def\content{
\begin{questions}
  \question An airplane is flying with a horizontal component velocity of 247 m/s and a vertical component velocity of 166 m/s. What is the resultant velocity of the airplane, and at what angle is it flying?  \emph{Make sure to draw a triangle.} \vspace{12em} 
  \question To throw a runner out at home plate, a player throws a baseball with a velocity of 41 m/s at a 36-degree angle. What are the horizontal and vertical components of his throw? \emph{Make sure to draw a triangle.} \vspace{12em}  

  \hrule 

  \subsection*{Equations}

  \begin{minipage}{2in}
    \begin{tikzpicture}[
        every node/.append style={font=\large},
        component/.append style={
          dashed,
          -{Stealth}
        },
        resultant/.append style={
          ultra thick,
          -{Stealth}
        },
        scale=0.8
    ]
      \draw[resultant] 
                (0,0) -- (4.5,3) 
                  node[midway,anchor=south east]{\bf $v_R$};
      \draw[component] 
                (0,0) 
                -- (4.5,0) 
                  node[midway,anchor=north]{$v_x$} ;
      \draw[component] 
                (4.5,0)
                -- (4.5,3) 
                  node[midway, anchor=west]{$v_y$} ;
      \draw[-Stealth] (1,0) node[anchor=south west]{$\theta$}
                arc[
                  start angle = 0,
                  end angle = 31, 
                  radius = 1
                ];
    \end{tikzpicture}
  \end{minipage}
  %
  \begin{minipage}{1.5in}
      \begin{align*}
        \theta  &= \tan^{-1} \left( v_y / v_x \right)\\
        v_x^2 + v_y^2 &= v_R^2                       \\[1em]
        v_x     &= v_R \cos \left( \theta \right)    \\
        v_y     &= v_R \sin \left( \theta \right)
      \end{align*}
  \end{minipage}
\end{questions}
}
