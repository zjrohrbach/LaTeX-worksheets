\begin{questions}


\question
  A \emph{converging} lens has a focal length of 20.0~m.  It is 60.0~m away from an antelope.

  \begin{parts}
    \part What would be the distance of the image from the lens? \vs
    \part What is the magnification of the image? \vs
  \end{parts}


\question
  You place a penny under a converging lens of focal length 75~mm.  You hold the lens 50~mm away from the penny.  

  \begin{parts}
    \part What would be the distance of the image from the lens? \vs
    \part What is the magnification of the lens? \vs
    \part if the diameter (height) of the penny is 19 mm, what will be the diameter (height) of the image? \vs
  \end{parts}
