\begin{questions}
  \question
    A student is riding a bicycle at 10~m/s toward a stationary factory whistle that is blowing at a frequency of 800~Hz. What frequency does the student perceive? \vs

  \question
    A police car with a \SI{600}{\hertz} siren is chasing a speeding car. The police car moves at \SI{40}{\meter\per\second}, while the speeding car moves away from the police at \SI{35}{\meter\per\second}. What frequency does the driver of the speeding car hear? \vs


  \question \emph{Challenge:}
    A bat uses echolocation to orient itself to its surroundings.  One such bat flies at \SI{12}{m/s} directly toward a large, stationary wall while emitting an ultrasonic pulse at a frequency of \SI{30000}{Hz}.
    
    \begin{parts}
      \part What is the frequency of the sound waves as they reach the stationary wall? \vs 
      \part What frequency does the bat hear when those sound waves reflect off the wall and return to the bat? \vs 

    \end{parts}

\end{questions}
