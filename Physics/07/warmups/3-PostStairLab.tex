\begin{questions}
  \question While climbing the stairs…
  \begin{parts}

      \part …did you output more power if you were going fast or slow?
  \end{parts}
  \question While climbing at a constant speed…
  \begin{parts}
    \part …did you do more work if you climbed the short stairs or tall stairs?
    \part …did you output more power if you climbed the short stairs or tall stairs?
  \end{parts}
\end{questions}
}
