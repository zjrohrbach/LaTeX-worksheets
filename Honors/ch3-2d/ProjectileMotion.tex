\begin{center}
  \large $\vec{v} = \vec{v}_0 + \vec{a}t$ \hfill 
  $\vec{x}=\vec{x}_0+\vec{v}_0 t + \frac{1}{2}\vec{a}t^2$ \hfill
  $\vec{v}^2 = \vec{v}_0^2 + 2 \vec{a}\cdot(\vec{x}-\vec{x_0})$
\end{center}
