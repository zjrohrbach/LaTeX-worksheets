\section*{Rotational Kinematic Equations}

\begin{multicols}{3}
  \begin{center}
      \emph{``Old Faithful''} 
      %
      \begin{align*}
        v = v_0 + a t
      \end{align*}

      \columnbreak
      
      \emph{``The Big Chalupa''} 
      %
      \begin{align*}
        x = x_0 + v_0t + \frac{1}{2}at^2
      \end{align*}

      \columnbreak
      
      \emph{``Ain't Got No Time''}
      %
      \begin{align*}
        v^2 = v_0^2 + 2a \left( x - x_0 \right)
      \end{align*}

  \end{center}
\end{multicols}
